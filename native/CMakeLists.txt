cmake_minimum_required(VERSION 3.20)
project(WsmBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "WsmBridge requires Windows x64")

target_include_directories(wsm_il2cpp PUBLIC include)
add_library(wsm_ingame_localization STATIC src/ingame_localization.cpp)

add_library(wsm_ingame_ui STATIC src/ingame_ui.cpp)
target_include_directories(wsm_ingame_ui PUBLIC include)
target_compile_definitions(wsm_ingame_ui PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(wsm_ingame_ui PUBLIC wsm_il2cpp wsm_ingame_localization user32 comctl32)
add_library(wsm_bridge SHARED src/bridge.cpp src/main_probe.cpp src/ingame_loop.cpp)
target_link_libraries(wsm_bridge PRIVATE wsm_il2cpp wsm_window_hook wsm_ingame_runtime wsm_ingame_ui wsm_player_observer)
