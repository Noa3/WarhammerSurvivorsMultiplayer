cmake_minimum_required(VERSION 3.20)
project(WsmBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "WsmBridge requires Windows x64")
endif()
add_library(wsm_il2cpp STATIC src/il2cpp_api.cpp)
target_include_directories(wsm_il2cpp PUBLIC include)
add_library(wsm_window_hook STATIC src/window_hook.cpp)
target_include_directories(wsm_window_hook PUBLIC include)
target_link_libraries(wsm_window_hook PUBLIC user32)
add_library(wsm_bridge SHARED src/bridge.cpp src/main_probe.cpp)
target_link_libraries(wsm_bridge PRIVATE wsm_il2cpp wsm_window_hook)
set_target_properties(wsm_bridge PROPERTIES OUTPUT_NAME "Wsm.Bridge")
include(CTest)
if(BUILD_TESTING)
    add_executable(wsm_bridge_fixture ../tests/native/fixture_process.cpp)
    set_target_properties(wsm_bridge_fixture PROPERTIES OUTPUT_NAME "Wsm.BridgeFixture")
    add_library(wsm_load_fixture SHARED ../tests/native/load_fixture.cpp)
    set_target_properties(wsm_load_fixture PROPERTIES OUTPUT_NAME "Wsm.LoadFixture")
    add_executable(bridge_test ../tests/native/bridge_test.cpp)
    add_test(NAME bridge_load COMMAND bridge_test "$<TARGET_FILE:wsm_bridge>")
    add_library(il2cpp_fixture SHARED ../tests/native/fixture.cpp)
    target_include_directories(il2cpp_fixture PRIVATE include)
    add_executable(api_test ../tests/native/api_test.cpp)
    target_link_libraries(api_test PRIVATE wsm_il2cpp)
    add_test(NAME api_contract COMMAND api_test "$<TARGET_FILE:il2cpp_fixture>")
    add_executable(window_hook_test ../tests/native/window_hook_test.cpp)
    target_link_libraries(window_hook_test PRIVATE wsm_window_hook)
    add_test(NAME window_hook COMMAND window_hook_test)
    add_test(NAME window_hook_timeout COMMAND window_hook_test timeout)
endif()
if(MSVC)
    foreach(target wsm_il2cpp wsm_bridge)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    endforeach()
endif()
