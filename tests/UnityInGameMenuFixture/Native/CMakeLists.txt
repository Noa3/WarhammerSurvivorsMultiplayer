cmake_minimum_required(VERSION 3.20)
project(InGameMenuFixture LANGUAGES CXX)
add_library(ingame_ui_fixture SHARED fixture.cpp ../../../native/src/ingame_localization.cpp ../../../native/src/ingame_ui.cpp ../../../native/src/il2cpp_api.cpp)
target_include_directories(ingame_ui_fixture PRIVATE ../../../native/include)
target_link_libraries(ingame_ui_fixture PRIVATE comctl32)
target_compile_features(ingame_ui_fixture PRIVATE cxx_std_17)
target_compile_definitions(ingame_ui_fixture PRIVATE WSM_INGAME_UI_TESTING WIN32_LEAN_AND_MEAN NOMINMAX)
# Also compile the production translation unit without test-only entry points.
add_library(ingame_ui_production OBJECT ../../../native/src/ingame_ui.cpp)
target_include_directories(ingame_ui_production PRIVATE ../../../native/include)
target_compile_features(ingame_ui_production PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(ingame_ui_fixture PRIVATE /utf-8)
    target_compile_options(ingame_ui_production PRIVATE /utf-8)
endif()
target_compile_definitions(ingame_ui_production PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
